  mode = 'fly';
  const p = toWorld(s.e, s.n);
  yaw = -THREE.MathUtils.degToRad(s.heading);

  resize();
}

function fly(dt: number) {
  const f = (keys.has('KeyW') || keys.has('ArrowUp') ? 1 : 0) - (keys.has('KeyS') || keys.has('ArrowDown') ? 1 : 0) + flyTouch;
  const r = (keys.has('KeyD') || keys.has('ArrowRight') ? 1 : 0) - (keys.has('KeyA') || keys.has('ArrowLeft') ? 1 : 0);
  const u = (keys.has('KeyE') || keys.has('Space') ? 1 : 0) - (keys.has('KeyQ') || keys.has('KeyC') ? 1 : 0);
  if (f || r || u) pendingShot = null;
  const v = flySpeed * (keys.has('ShiftLeft') || keys.has('ShiftRight') ? 5 : 1) * dt;
  camera.rotation.set(pitch, yaw, 0);
  const dir = new THREE.Vector3(0, 0, -1).applyEuler(camera.rotation);
  const right = new THREE.Vector3(1, 0, 0).applyEuler(camera.rotation);
  camera.position.addScaledVector(dir, f * v).addScaledVector(right, r * v);
  camera.position.y += u * v;
  const g = terrain.heightAt(camera.position.x, camera.position.z);
  if (g !== null) {
    const floor = Math.max(g, SEA_LEVEL) + 1.2;
    if (pendingShot) camera.position.y = Math.max(g, SEA_LEVEL) + pendingShot.eye!;
    else if (camera.position.y < floor) camera.position.y = floor;
  }
}

// The chase camera trails the car's direction of travel and never dips under the ground.
const camPos = new THREE.Vector3(), camLook = new THREE.Vector3();
let camHeading = 0, camReady = false;
function chase(dt: number) {
  if (view === 'seat') {
    car.driverEye.getWorldPosition(camera.position);
    camera.rotation.set(vehicle.pitch * 0.8, -vehicle.heading, 0);
    if (camera.fov !== 68) { camera.fov = 68; camera.updateProjectionMatrix(); }
    return;
  }
  if (camera.fov !== 58) { camera.fov = 58; camera.updateProjectionMatrix(); }
  const moving = Math.hypot(vehicle.vx, vehicle.vz) > 2 && vehicle.speed > 0;
  const want = moving ? Math.atan2(vehicle.vx, -vehicle.vz) : vehicle.heading;
  if (!camReady) camHeading = want;
  camHeading += Math.atan2(Math.sin(want - camHeading), Math.cos(want - camHeading)) * Math.min(1, 3.5 * dt);
  const back = vehicleSpec.chase.back + Math.min(Math.abs(vehicle.speed) * vehicleSpec.chase.speedBack, vehicleSpec.chase.maxSpeedBack);
  const tx = vehicle.x - Math.sin(camHeading) * back, tz = vehicle.z + Math.cos(camHeading) * back;
  let ty = vehicle.y + vehicleSpec.chase.height;
  const g = terrain.surfaceAt(tx, tz);
  if (g) ty = Math.max(ty, g.ground + g.canopy * 0.5 + 1.3);
  const k = camReady ? Math.min(1, 7 * dt) : 1;
  camPos.lerp(new THREE.Vector3(tx, ty, tz), k);
  camLook.lerp(new THREE.Vector3(vehicle.x + Math.sin(vehicle.heading) * vehicleSpec.chase.lookAhead, vehicle.y + vehicleSpec.chase.lookHeight, vehicle.z - Math.cos(vehicle.heading) * vehicleSpec.chase.lookAhead), k);
  camera.position.copy(camPos);
  camera.lookAt(camLook);
  camReady = true;
}

function frame(now: number) {
  const clockDt = Math.max(0, (now - last) / 1000);
  const realDt = Math.min(clockDt, 0.1);
  const dt = shift.ended || !started || endings.busy || document.hidden ? 0 : realDt;
  if (!started) camReady = false;   // behind the title card the camera snaps to the car as the ground arrives
  if (shift.ended) vehicle.vx = vehicle.vz = vehicle.speed = 0;
  last = now;
  fps += (1 / Math.max(realDt, 1e-3) - fps) * 0.05;
  if (endings.busy) {
    drawEnding(realDt); requestAnimationFrame(frame); return;
  }
  syncRoadScenePause();
  if (mode === 'fly') { yardJob?.judge.cancel(); fishing?.judge.cancel(); }
  const loaded = terrain.pending === 0;
  // The one measurement of the picture at a first visit (picture.ts): only frames of real play count.
  if (picture?.measuring) picture.frame(clockDt, !started || !loaded || document.hidden || shift.ended || interiors.busy);
  if (lawnDemo?.prepare(terrain, vehicle)) { camReady = false; drove = 0; }
  vehicle.damageActive = mailMode && started && !shift.ended && !interiors.busy && mode === 'drive' && !demo && !waitingToDrive;
  car.setWindowDown?.(shift.damage.windowStuck);
  radio?.setWind(shift.damage.windowStuck, mode === 'drive' ? vehicle.speed : 0);
  if (interiors.busy) {
    radio?.update(false, true);
    pad?.show(!shift.ended);
    if (shift.updateIndoors(started && !shift.ended ? clockDt : 0)) { interiors.closeForDay(); shift.end(); }
    else {
      interiors.update(dt, driveInput(), vehicle, false);
      interiors.render(renderer);
    }
    // Interior scenes have no loading assets. The frozen world resumes on the next frame.
    quiet++; document.title = quiet > 20 ? 'settled' : 'loading';
    mailRequested = photoRequested = false;
    if (!interiors.busy) { keys.clear(); touch.steer = touch.throttle = 0; camReady = false; }
    requestAnimationFrame(frame); return;
  }
  interiors.update(realDt, driveInput(), vehicle, started && !shift.ended && mode === 'drive');
  pad?.show(mode === 'drive' && !shift.ended && !placesUI?.busy);
  if (mode === 'drive') {
    const input = demo && drove < demo ? autopilot(vehicle, roads, params.get('throw') === '1' ? 8 : 16) : demo ? { throttle: Math.abs(vehicle.speed) > 0.15 ? -Math.sign(vehicle.speed) : 0, steer: 0, handbrake: true } : driveInput();
    if (demo && drove < demo && params.get('knock') === '1') {
      const road = roads.nearest(vehicle.x, vehicle.z, 40, (r) => r.kind <= 3);
      if (road) {
        let heading = road.heading;
        if (Math.cos(heading - vehicle.heading) < 0) heading += Math.PI;
        const rx = Math.cos(heading), rz = Math.sin(heading);
        let shoulder = road.road.halfWidth + .7, nearest = 55;
        // Follow the right shoulder, then line up on an upright post ahead.
        if (props) for (const box of props.mail) {
          if (box.motion.down) continue;
          const dx = box.x - vehicle.x, dz = box.z - vehicle.z;
          const ahead = dx * Math.sin(heading) - dz * Math.cos(heading);
          const across = (box.x - road.x) * rx + (box.z - road.z) * rz;
          if (ahead > -2 && ahead < nearest && across > 0 && across < road.road.halfWidth + 4) { nearest = ahead; shoulder = across; }
        }
        const side = (vehicle.x - road.x) * rx + (vehicle.z - road.z) * rz - shoulder;
        const err = Math.atan2(Math.sin(heading - vehicle.heading), Math.cos(heading - vehicle.heading));
        input.steer = THREE.MathUtils.clamp(err * 1.8 - side * .3, -1, 1);
      }
    }
    if (mailMode && params.has('lawn') && demo && drove < demo) { input.throttle = 1; input.steer = 0; input.handbrake = false; }
    if (lawnDemo && !lawnDemo.driving) { input.throttle = 0; input.handbrake = true; }

      input.handbrake = true;

  terrain.update(camera);

  landmarks?.update(camera);
  {
    placesUI?.update(vehicle, started && mode === 'drive');
    const playing = mode === 'drive' && started && !shift.ended;
    if (params.get('break') === '1' && relief.breaks === 0 && (!demo || drove >= demo)) relief.request();
    relief.update(dt, playing ? clockDt * CLOCK_FACTOR : 0, vehicle, playing && (!demo || reliefTest > 0));
    if (miniMap) {
      miniMap.show(playing);
      mapTimer -= realDt;
      if (mapTimer <= 0) { mapTimer = 0.25; refreshMap(); }
      miniMap.update(realDt, vehicle.x, vehicle.z, vehicle.heading, mapStops, mapMarks, mapCars);
    }
    radio?.update(playing, false);
  }
  roads.update(camera.position);
  edge?.update(camera.position);
  mineSite?.update(camera.position);
  fishing?.update(dt, camera.position);
  roadScene?.update(camera);
  props?.update(camera, dt);
  if (loaded) shiftReady = true;
  if (shiftReady) shift.update(shift.ended || !started ? 0 : clockDt, vehicle);
  if (sunShadows) {
    // The shadow box follows the car; with the free camera it sits a little ahead of the lens.
    let reach = 0;
    if (mode === 'drive') shadowFocus.set(vehicle.x, vehicle.y, vehicle.z);
    else {
      // Walk down the line of sight to the ground, in coarse steps: that is what the picture is of.
      shadowDir.set(0, 0, -1).applyQuaternion(camera.quaternion);
      for (reach = 20; reach < 1400; reach *= 1.35) {
        shadowFocus.copy(camera.position).addScaledVector(shadowDir, reach);
        const g = terrain.heightAt(shadowFocus.x, shadowFocus.z);
        if (g !== null && shadowFocus.y <= g) { shadowFocus.y = g; break; }
      }
      if (reach >= 1400) {
        // A high, shallow look meets no ground within reach. Then the box sits on the ground ahead of the camera,
        // as far ahead as the camera is high, and is sized for that height.
        const under = terrain.heightAt(camera.position.x, camera.position.z) ?? SEA_LEVEL;
        const height = Math.max(20, camera.position.y - under), flat = Math.hypot(shadowDir.x, shadowDir.z);
        // A look that is nearly straight down has no "ahead": the box then stays under the camera.
        const ahead = flat > 0.2 ? height / flat : 0;
        shadowFocus.set(camera.position.x + shadowDir.x * ahead, 0, camera.position.z + shadowDir.z * ahead);
        shadowFocus.y = terrain.heightAt(shadowFocus.x, shadowFocus.z) ?? under;
        reach = Math.min(1400, height * 1.5);
      }
    }
    sunShadows.update(realDt, shadowFocus, reach);
  }
  grass?.update(realDt, camera, vehicle.x, vehicle.z, vehicle.heading, vehicleSpec);
  // LOW water is a backdrop. Enable site clipping only while the wharf/boat is visible.
  renderer.localClippingEnabled = LOW && !!fishing?.group.visible;
  multiplayer?.update(realDt, camera, dt > 0 && mode === 'drive');
  waterReflections?.update(camera);
  try { renderer.render(scene, camera); } finally { waterReflections?.settle(); }
  waterReflections?.afterRender(camera);
  // A photo is evidence of what the driver saw: the free camera takes none.
  if (photoRequested) { if (mode === 'drive') shift.takePhoto(vehicle, camera); photoRequested = false; }
  if (params.get('photo') === '1' && demo && !demoPhoto && loaded && mode === 'drive') {
    // Only press the shutter once something is in range and inside the camera or front cone.
    for (const s of shift.evidence.subjects.values()) if (s.active && Math.hypot(s.x - vehicle.x, s.z - vehicle.z) <= 43) {
      if (shift.takePhoto(vehicle, camera)) demoPhoto = true;
      break;
    }
  }
  if (params.get('report') === '1' && loaded && (!demo || drove >= demo) && Math.abs(vehicle.speed) < .15
    && !relief.busy && (params.get('break') !== '1' || relief.breaks > 0)) shift.end();

  // A page is "settled" when nothing has been loading for a while; screenshots wait for this.
  const waiting = !loaded || (grass?.pending ?? 0) > 0 || (trees?.pending ?? 0) > 0 || (mode === 'fly' && pendingShot !== null && terrain.heightAt(camera.position.x, camera.position.z) === null)
    || (mode === 'drive' && demo > 0 && (drove < demo || Math.abs(vehicle.speed) > 0.15)) || relief.busy;
  quiet = waiting ? 0 : quiet + 1;
  document.title = quiet > 20 ? 'settled' : 'loading';
  if (mailButton) mailButton.hidden = mode !== 'drive' || shift.ended;
  if (mailScore && !bare && mailScore.textContent !== mail.tally) mailScore.textContent = mail.tally;
  if (!bare) {
    // Behind the title card the readout said "0 km/h off the map" while the land was still loading: keep it out of sight.
    hud.style.visibility = started ? '' : 'hidden';
    if (mode === 'drive') {
      hud.textContent = `${Math.abs(vehicle.speed * 3.6).toFixed(0)} km/h   ${vehicle.roadName ?? (yardSite?.surface(vehicle.x, vehicle.z) ? 'works yard' : 'off the map')}   ${vehicle.surfaceName}${vehicle.airborne ? '   AIR' : ''}` +
        (touchDevice ? `   ${fps.toFixed(0)} fps`
          : !mailMode ? exploreHelp : `\nWASD/arrows: drive · space: brake · M/E: mail · P: photo · Enter: end shift\nH: horn · R: road · V: seat · F: fly · ${fps.toFixed(0)} fps`);
    } else {
      const e = camera.position.x + VE, n = VN - camera.position.z;
      hud.textContent = `E ${e.toFixed(0)}  N ${n.toFixed(0)}  h ${camera.position.y.toFixed(1)} m   hills x${EXAG}\n` +
        `sun az ${sun.az.toFixed(0)} el ${sun.el.toFixed(0)}  tiles ${terrain.drawn}  load ${terrain.pending}  ${fps.toFixed(0)} fps\n` +
        `drag: look · WASD: fly · ${mailMode ? 'Q/E' : 'Q/space'}: down/up · wheel: speed ${flySpeed.toFixed(0)} m/s${SHOTS.length ? ` · 1-${SHOTS.length}: views` : ''} · F: drive`;
    }
    if (mailMode && mode === 'drive' && !touchDevice) hud.textContent += `\n${shift.done}/${shift.stops} stops · ${mail.delivered} pieces delivered · ${mail.lawn + mail.ditch} on ground`;
  }
  if (vehicle.atEdge && mode === 'drive' && !bare) hud.textContent += !mailMode ? '\nThe map ends here.' : AREA.route ? '\nEnd of your route. Somebody else delivers past here.' : '\nEnd of this map.';
  if (mailMode && params.has('lawn') && !bare) hud.textContent += `\nTyre strips: ${shift.ruts.count} · Lawns damaged: ${shift.ruts.lawns}`;
  requestAnimationFrame(frame);
}
